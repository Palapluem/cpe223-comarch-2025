\subsection*{Part 2: Short Answer Solutions}
\begin{enumerate}
    \item \textbf{ALU}: หน่วยคำนวณทางคณิตศาสตร์และตรรกะ (Arithmetic and Logic Unit)
    \item \textbf{IC}: วงจรรวม (Integrated Circuit) ชิปที่รวมวงจรอิเล็กทรอนิกส์ไว้ในสารกึ่งตัวนำ
    \item \textbf{PC}: เก็บ Address ของคำสั่งถัดไปที่จะถูก Fetch มาทำงาน
    \item \textbf{IR}: เก็บ Instruction ปัจจุบันที่กำลังถูก Decode/Execute อยู่
    \item \textbf{SP}: ชี้ตำแหน่งบนสุดของ Stack (Top of Stack) ใช้เก็บข้อมูลชั่วคราวหรือ Return Address
    \item \textbf{RISC}: สถาปัตยกรรมคอมพิวเตอร์ที่เน้นชุดคำสั่งขนาดเล็กและเรียบง่าย (Reduced Instruction Set Computer)
    \item \textbf{Direct Addressing}: การระบุ Memory Address ลงไปตรง ๆ ในตัวคำสั่งเลย (เช่น LOAD R1, 1000)
    \item \textbf{LOAD R2, 4(R1)}: เอาค่าใน R1 บวกกับ 4 ได้ Address ปลายทาง แล้วไป Load ข่าจาก Address นั้นมาใส่ R2 (Indexed/Displacement Mode)
    \item \textbf{Advantage of Interrupt}: CPU ไม่ต้องเสียเวลาวนลูปตรวจสอบสถานะ (Polling) สามารถทำงานอื่นไปได้เรื่อย ๆ จนกว่าอุปกรณ์จะแจ้งเตือน
    \item \textbf{Code Explanation}: ลูปนี้จะลดค่า R1 ทีละ 1 และวนซ้ำตราบเท่าที่ R1 $\ge$ 0 (วนลูปตามจำนวนรอบที่กำหนดใน R1 หรือจนกว่า R1 จะติดลบ)
\end{enumerate}

\subsection*{Part 3: Problem Solving Solutions}

\textbf{1. Memory \& Stack Content Step-by-Step Trace}
\\ \textit{สมมติฐาน: การบวก/ลบ SP ด้วย \#1 หมายถึงการเลื่อนไป Address ถัดไปตามขนาด Word (2 bytes) ในระบบจำลองนี้}

\begin{enumerate}[label=\arabic*.]
    \item \textbf{Initial State}: SP = 1000, Mem[1000]=1002, Mem[1002]=1004, ...
    \item \texttt{Load R0, \#1004}: R0 = 1004
    \item \texttt{Load R1, \#1006}: R1 = 1006
    \item \texttt{Load R2, (SP)}: อ่าน Mem[1000] $\to$ R2 = 1002
    \item \texttt{Add SP, \#1}: SP เลื่อนลง $\to$ SP = 1002 (Address ถัดไป)
    \item \texttt{Load R3, (SP)}: อ่าน Mem[1002] $\to$ R3 = 1004
    \item \texttt{Add SP, \#1}: SP เลื่อนลง $\to$ SP = 1004
    \item \texttt{Subtract SP, \#1}: SP ย้อนกลับ $\to$ SP = 1002
    \item \texttt{Store (SP), (R0)+2}: 
    \\ - ปลายทาง: (SP) คือ Mem[1002]
    \\ - ต้นทาง: ใช้ R0 (1004) เป็น Address $\to$ อ่าน Mem[1004] ได้ค่า \textbf{1006}
    \\ - หลังอ่าน R0 บวอกเพิ่ม 2 $\to$ R0 = 1006 (Post-increment)
    \\ - \textbf{ผลลัพธ์}: เขียนค่า 1006 ทับลงใน Mem[1002] (ค่าเดิมคือ 1004 หายไป)
    \item \texttt{Add R2, ((R1))}:
    \\ - R1 = 1006
    \\ - Value = Mem[Mem[1006]] = Mem[1008] = 1010
    \\ - R2 = R2 + 1010 = 1002 + 1010 = \textbf{2012}
    \item \texttt{Add R2, (R3)+4}:
    \\ - R3 ปัจจุบันคือ 1004 (โหลดไว้ตั้งแต่ข้อ 6 ก่อน Mem[1002] จะเปลี่ยน และ R3 ไม่ได้ถูกแก้)
    \\ - อ่าน Mem[R3] = Mem[1004] = 1006
    \\ - R2 = R2 + 1006 = 2012 + 1006 = \textbf{3018}
    \\ - R3 บวกเพิ่ม 4 $\to$ R3 = 1008
    \item \texttt{Store R1, R2}:
    \\ - เขียนค่า R2 (3018) ลงใน Address ที่ R1 ชี้ (1006)
    \\ - \textbf{ผลลัพธ์}: Mem[1006] เปลี่ยนเป็น 3018
\end{enumerate}

\textbf{2. C to Assembly Translation}
\\ \textbf{Logic Idea}: ใช้ R0 แทน N (Index), R1 แทน SUM, R2 เก็บ Base Address ของ Array

\begin{minipage}{0.9\textwidth}
\begin{verbatim}
       ; Initialization
       LOAD  R0, #9           ; N = 9
       LOAD  R1, #0           ; SUM = 0
       LOAD  R2, #NUM_BASE    ; R2 points to start of num[] array
       
LOOP:  
       ; Check Loop Condition (N >= 0)
       CMP   R0, #0
       BLT   END_LOOP         ; If N < 0, Exit Loop
       
       ; Load num[N]
       ; Assuming integer is 1 word (offset N)
       ; Calculate address = R2 + R0
       ADD   R3, R2, R0       ; R3 = Address of num[N] (Simplified)
       LOAD  R4, (R3)         ; R4 = num[N]
       
       ; Check if (num[N] % 4) == 0
       ; Modulo 4 is equivalent to checking last 2 bits (AND 3)
       AND   R5, R4, #3       ; R5 = num[N] & 3
       CMP   R5, #0
       BNE   SKIP_SUM         ; If not 0, it's not divisible by 4
       
       ; SUM += num[N]
       ADD   R1, R1, R4       ; SUM = SUM + num[N]
       
SKIP_SUM:
       ; N--
       SUB   R0, #1           ; Decrement N
       BRA   LOOP             ; Jump back to start
       
END_LOOP:
       ; Finished
\end{verbatim}
\end{minipage}
